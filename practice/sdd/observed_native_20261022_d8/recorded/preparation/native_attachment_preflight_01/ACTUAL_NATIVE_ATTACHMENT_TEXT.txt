71: SHA-256: 14d186a327a418dddb405144bfefb5810ed27e626b981424c9ad3b66742e8b32
72: ```python
73: """A recommendation routing service. This module never authorizes fund movement."""
74: from dataclasses import dataclass
75: from decimal import Decimal, InvalidOperation
76: 
77: THRESHOLD_EUR = Decimal("5000.00")
78: POLICY_VERSION = "BR-001/v1"
79: 
80: @dataclass(frozen=True)
81: class RoutingDecision:
82:     route: str
83:     reason: str
84:     policy_version: str
85: 
86: def assess(amount_eur: str) -> RoutingDecision:
87:     """Choose a processing queue. Not a payment authorisation."""
88:     try:
89:         amount = Decimal(amount_eur)
90:     except (InvalidOperation, ValueError, TypeError):
91:         return RoutingDecision("REJECTED", "INVALID_AMOUNT", POLICY_VERSION)
92:     if not amount.is_finite() or amount <= 0:
93:         return RoutingDecision("REJECTED", "INVALID_AMOUNT", POLICY_VERSION)
94:     if amount < THRESHOLD_EUR:
95:         return RoutingDecision("STANDARD_QUEUE", "BELOW_LIMIT", POLICY_VERSION)
96:     return RoutingDecision("HUMAN_REVIEW", "AT_OR_ABOVE_LIMIT", POLICY_VERSION)
