```
"""Existing JSON presentation; add CSV after the same selection boundary."""
from selection import selected_orders, public_order

def orders_response(rows, actor, format="json", status=None):
    selected = selected_orders(rows, actor, status)
    public = [public_order(row) for row in selected]
    if format == "json":
        return {"content_type": "application/json", "body": public}
    if format == "csv":
        raise NotImplementedError("CSV presentation requested but not implemented")
    raise ValueError("unsupported format")

```

## selection.py
```
"""Existing owner-controlled selection. Do not broaden access for CSV."""
def selected_orders(rows, actor, status=None):
    return [row for row in rows
            if row["owner"] == actor and (status is None or row["status"] == status)]

def public_order(row):
    return {field: row[field] for field in
            ("order_id", "status", "customer", "amount_eur", "note")}

```

```
"""Local function demonstration, not an authenticated HTTP deployment."""
import argparse
import json
from pathlib import Path
from api import orders_response

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--format", choices=["json", "csv"], default="json")
    parser.add_argument("--actor", default="alice")
    parser.add_argument("--status", default=None)
    args = parser.parse_args()
    rows = json.loads((Path(__file__).parent / "orders.json").read_text())
    result = orders_response(rows, args.actor, args.format, args.status)
    if args.format == "json":
        print(json.dumps(result["body"], indent=2))
    else:
        print(result["body"], end="")

if __name__ == "__main__":
    main()
```
